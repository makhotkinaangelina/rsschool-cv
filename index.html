Codewars: Human Readable Duration Format, 4 kyu 

function formatDuration(seconds) {
  const formats = [
    { word: " year", value: 31536000, amount: 0 },
    { word: " day", value: 86400, amount: 0 },
    { word: " hour", value: 3600, amount: 0 },
    { word: " minute", value: 60, amount: 0 },
    { word: " second", value: 1, amount: 0 }
  ];

  let filledFormats = [];
  let result = "";

  for (let i = 0; i < formats.length; i++) {
    const amount = Math.floor(seconds / formats[i].value);
    seconds %= formats[i].value;

    formats[i].amount = amount;

    if (amount) {
      filledFormats.push(i);

      if (amount > 1) {
        formats[i].word += "s";
      }
    }
  }

  switch (filledFormats.length) {
    case 0:
      result = "now";
      break;
    case 1:
      result = `${formats[filledFormats[0]].amount}${formats[filledFormats[0]].word}`;
      break;
    default:
      result = filledFormats
        .map(index => `${formats[index].amount}${formats[index].word}`)
        .join(", ")
        .replace(/, ([^,]*)$/, " and $1");
  }

  return result;
}
